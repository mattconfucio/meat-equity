    <script>
        /* === CALCULADORA DE CASCATA BIOLÓGICA === */
        
        function calculateCascade() {
            const initialKg = parseFloat(document.getElementById('inputKg').value);
            if (isNaN(initialKg) || initialKg <= 0) return;

            // Arrays to hold the state of cattle in each cycle phase.
            // state[0] = Ciclo 1 (100% Cotista)
            // state[1] = Ciclo 2 (Partilha)
            // state[2] = Ciclo 3 (Partilha), etc.
            let state = [initialKg, 0, 0, 0, 0, 0, 0, 0];
            let historyTotals = [];

            // Run 7 cycles (1150 days)
            for (let cycle = 1; cycle <= 7; cycle++) {
                // 1. Calculate weight gain (50% per phase)
                let engordaC1 = state[0] * 0.5;
                let engordaPartilha = 0;
                for (let j = 1; j < state.length; j++) {
                    engordaPartilha += state[j] * 0.5;
                }

                // 2. Distribute gains according to Meat Equity Rules
                // Cotista gets 100% of C1 gain + 50% of Partilha gain. This is reinvested into a new C1.
                let cotistaReinvests = engordaC1 + (engordaPartilha / 2);

                // 3. Shift the cascades (Animals move to next phase)
                let nextState = [0, 0, 0, 0, 0, 0, 0, 0];
                nextState[0] = cotistaReinvests; // New animals go to C1
                nextState[1] = state[0];         // C1 moves to C2
                nextState[2] = state[1];         // C2 moves to C3
                nextState[3] = state[2];
                nextState[4] = state[3];
                nextState[5] = state[4];
                nextState[6] = state[5];
                nextState[7] = state[6] + state[7]; // C7+ accumulates

                state = nextState;
                
                // 4. Calculate total Patrimony for the Cotista
                let currentTotal = state.reduce((a, b) => a + b, 0);
                historyTotals.push(currentTotal);
            }

            updateUI(initialKg, historyTotals);
        }

        function updateUI(initialKg, historyTotals) {
            const finalTotal = historyTotals[historyTotals.length - 1];
            const multiplier = (finalTotal / initialKg).toFixed(1);

            // Animate Numbers
            animateValue("finalResult", 0, finalTotal, 1000, " kg");
            document.getElementById('multiplierResult').innerText = multiplier + "x";

            // Draw Chart
            const chartContainer = document.getElementById('chartContainer');
            chartContainer.innerHTML = ''; // Clear old

            const maxVal = finalTotal;

            historyTotals.forEach((val, index) => {
                const heightPct = (val / maxVal) * 100;
                
                // Determine color based on index (last one is gold)
                const isLast = index === historyTotals.length - 1;
                const bgColor = isLast ? 'bg-brand-gold' : 'bg-brand-forest';
                
                const barHtml = `
                    <div class="relative flex flex-col justify-end items-center flex-1 h-full group">
                        <div class="absolute -top-8 text-xs font-bold text-gray-500 opacity-0 group-hover:opacity-100 transition whitespace-nowrap">
                            ${Math.round(val).toLocaleString('pt-BR')} kg
                        </div>
                        <div class="w-full mx-1 ${bgColor} rounded-t-sm bar-grow" style="height: 0%;" data-target="${heightPct}%"></div>
                    </div>
                `;
                chartContainer.innerHTML += barHtml;
            });

            // Trigger Animation
            setTimeout(() => {
                const bars = document.querySelectorAll('.bar-grow');
                bars.forEach(bar => {
                    bar.style.height = bar.getAttribute('data-target');
                });
            }, 50);
        }

        function animateValue(id, start, end, duration, suffix = "") {
            const obj = document.getElementById(id);
            let startTimestamp = null;
            const step = (timestamp) => {
                if (!startTimestamp) startTimestamp = timestamp;
                const progress = Math.min((timestamp - startTimestamp) / duration, 1);
                const currentVal = Math.floor(progress * (end - start) + start);
                // format with dots
                obj.innerHTML = currentVal.toLocaleString('pt-BR') + `<span class="text-xl text-gray-500 font-sans font-normal">${suffix}</span>`;
                if (progress < 1) {
                    window.requestAnimationFrame(step);
                }
            };
            window.requestAnimationFrame(step);
        }

        // Initialize chart on load
        window.onload = () => { calculateCascade(); };

        async function fetchGeminiAPI(prompt, systemPrompt) {
            const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] }
            };

            for (let i = 0; i < 5; i++) {
                try {
                    const response = await fetch(endpoint, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    
                    if (!response.ok) throw new Error('Network response was not ok');
                    
                    const data = await response.json();
                    return data.candidates?.[0]?.content?.parts?.[0]?.text || "Desculpe, não consegui formular a resposta. Pode tentar novamente?";
                } catch (error) {
                    if (i === 4) throw error;
                    await new Promise(resolve => setTimeout(resolve, Math.pow(2, i) * 1000));
                }
            }
        }

        async function sendChatMessage() {
            const inputField = document.getElementById('chat-input');
            const message = inputField.value.trim();
            if (!message) return;

            const chatMessages = document.getElementById('chat-messages');

            // 1. Add User Message
            const userMsgDiv = document.createElement('div');
            userMsgDiv.className = "bg-brand-forest text-white p-3 rounded-xl rounded-tr-none shadow-sm self-end max-w-[85%] text-sm";
            userMsgDiv.innerText = message;
            chatMessages.appendChild(userMsgDiv);
            inputField.value = '';
            chatMessages.scrollTop = chatMessages.scrollHeight;

            // 2. Add Loading Indicator
            const loaderDiv = document.createElement('div');
            loaderDiv.className = "bg-white p-4 rounded-xl rounded-tl-none shadow-sm border border-gray-100 self-start max-w-[85%]";
            loaderDiv.innerHTML = `
                <div class="loader-dots relative w-16 h-4">
                    <div class="absolute top-1 w-2 h-2 rounded-full bg-brand-gold"></div>
                    <div class="absolute top-1 w-2 h-2 rounded-full bg-brand-gold"></div>
                    <div class="absolute top-1 w-2 h-2 rounded-full bg-brand-gold"></div>
                    <div class="absolute top-1 w-2 h-2 rounded-full bg-brand-gold"></div>
                </div>
            `;
            chatMessages.appendChild(loaderDiv);
            chatMessages.scrollTop = chatMessages.scrollHeight;

            // 3. Call API
            const systemPrompt = `Você é o Consultor de Relações com Investidores do "Projeto Meat Equity", um investimento de elite em agropecuária (ativos biológicos).
            Regras do seu discurso:
            1. Seja polido, extremamente elegante, didático e profissional (nível M&A/Private Equity).
            2. O projeto é uma "Fábrica de Biomassa em Cascata".
            3. Fase de Aceleração (Ciclo 1): Custo é R$ 6,00 por kg inicial. A engorda é 100% do investidor (Foco em PAYBACK).
            4. Fase de Performance (Ciclos 2 em diante): Custo cai para R$ 1,25 por kg. A engorda (lucro) é dividida 50/50 entre Empresa e Investidor.
            5. O gado ganha 50% de peso a cada ciclo de 150 dias. 
            6. O lucro do investidor é SEMPRE reinvestido na Fase de Aceleração (novos animais).
            Responda de forma concisa e persuasiva em português.`;

            try {
                const responseText = await fetchGeminiAPI(message, systemPrompt);
                
                // Format response (convert markdown bold to HTML)
                const formattedText = responseText.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                
                // Replace loader with response
                loaderDiv.innerHTML = formattedText;
                loaderDiv.className = "bg-white p-4 rounded-xl rounded-tl-none shadow-sm border border-brand-gold/30 self-start max-w-[90%] text-gray-700 text-sm leading-relaxed";
            } catch (error) {
                loaderDiv.innerHTML = "Ocorreu um erro de comunicação com nossos servidores. Por favor, tente novamente.";
                loaderDiv.className = "bg-red-50 text-red-600 p-4 rounded-xl rounded-tl-none border border-red-200 self-start max-w-[85%] text-sm";
            }
            
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        // Allow 'Enter' key to send message
        document.getElementById('chat-input').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                sendChatMessage();
            }
        });
